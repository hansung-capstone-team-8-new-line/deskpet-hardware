cmake_minimum_required(VERSION 3.20)
project(deskpet_hardware VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(DESKPET_SDL "Build SDL2 screen renderer" ON)
set(_linux_default OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(_linux_default ON)
endif()
option(DESKPET_LINUX "Build ALSA and I2C drivers" ${_linux_default})
find_package(Threads REQUIRED)
add_library(deskpet_core
  src/config.cpp src/protocol.cpp src/audio.cpp src/motion.cpp src/pca9685.cpp
  src/display.cpp src/runtime.cpp src/server.cpp)
target_include_directories(deskpet_core PUBLIC include third_party)
target_link_libraries(deskpet_core PUBLIC Threads::Threads)
target_compile_options(deskpet_core PRIVATE -Wall -Wextra -Wpedantic)
if(DESKPET_SDL)
  find_package(SDL2 CONFIG REQUIRED)
  find_package(SDL2_ttf CONFIG QUIET)
  if(TARGET SDL2_ttf::SDL2_ttf)
    set(_sdl_ttf SDL2_ttf::SDL2_ttf)
  elseif(TARGET SDL2_ttf::SDL2_ttf-static)
    set(_sdl_ttf SDL2_ttf::SDL2_ttf-static)
  else()
    # Ubuntu 22.04's SDL_ttf supplies pkg-config metadata but no CMake config.
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2TTF REQUIRED IMPORTED_TARGET SDL2_ttf)
    set(_sdl_ttf PkgConfig::SDL2TTF)
  endif()
  target_link_libraries(deskpet_core PUBLIC SDL2::SDL2 ${_sdl_ttf})
  target_compile_definitions(deskpet_core PUBLIC DESKPET_SDL=1)
endif()
if(DESKPET_LINUX)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Linux hardware drivers require Linux")
  endif()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(ALSA REQUIRED IMPORTED_TARGET alsa)
  pkg_check_modules(SOXR REQUIRED IMPORTED_TARGET soxr)
  target_sources(deskpet_core PRIVATE src/audio_alsa.cpp src/motion_linux.cpp src/resampler.cpp)
  target_link_libraries(deskpet_core PUBLIC PkgConfig::ALSA PkgConfig::SOXR)
  target_compile_definitions(deskpet_core PUBLIC DESKPET_LINUX=1)
endif()
add_executable(deskpet-hardware src/main.cpp)
target_link_libraries(deskpet-hardware PRIVATE deskpet_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(deskpet-tests tests/core.cpp)
  target_link_libraries(deskpet-tests PRIVATE deskpet_core)
  add_test(NAME core COMMAND deskpet-tests)
  add_executable(deskpet-profile-tests tests/profile.cpp)
  target_link_libraries(deskpet-profile-tests PRIVATE deskpet_core)
  add_test(NAME current-profile COMMAND deskpet-profile-tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:deskpet-hardware>)
  set_tests_properties(integration PROPERTIES TIMEOUT 90)
endif()
install(TARGETS deskpet-hardware RUNTIME DESTINATION bin)
install(PROGRAMS tools/deskpetctl.py DESTINATION bin RENAME deskpetctl)
install(FILES config/jetson.json config/mock.json DESTINATION share/deskpet-hardware/config)
